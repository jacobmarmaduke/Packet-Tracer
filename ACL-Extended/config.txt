# Cisco Packet Tracer – ACL(Extended) Configuration

## Switch 1 Configuration

en
conf t

vlan 10
  exit

int range fa0/1-2
  switchport mode access
  switchport access vlan 10
exit

int g0/1
  switchport mode access
  switchport access vlan 10
exit


## Switch 2 Configuration

en
conf t

vlan 20
  exit

vlan 30
  exit

int fa0/1
  switchport mode access
  switchport access vlan 20
exit

int fa0/2
  switchport mode access
  switchport access vlan 30
exit

int g0/1
  switchport mode trunk
exit


## Router 1 Configuration


en
conf t

int g0/0
  ip address 192.168.10.1 255.255.255.0
  no shutdown
exit

(**Creating sub interfaces**)
int g0/1.20
  encapsulation dot1q 20
  ip address 192.168.20.1 255.255.255.0
  no shutdown
exit

int g0/1.30
  encapsulation dot1q 30
  ip address 192.168.30.1 255.255.255.0
  no shutdown
exit

(**Creating Extended ACL**)
ip access-list extended BLOCK-PC1-PC2-TO-PC4
  deny ip host 192.168.10.11 host 192.168.30.11
  deny ip host 192.168.10.12 host 192.168.30.11
  permit ip any any
exit

(**Apply ACL**)
int g0/0
  ip access-group BLOCK-PC1-PC2-TO-PC4 in
exit
